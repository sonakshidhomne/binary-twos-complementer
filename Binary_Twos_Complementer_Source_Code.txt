<div class="card">
<h2>7. How It Works</h2>
<ol>
<li>Check that the input contains only 0 and 1.</li>
<li>Start scanning from the rightmost bit.</li>
<li>Copy bits until the first 1 is reached.</li>
<li>Copy that 1 and move to state q1.</li>
<li>In q1, complement every remaining bit.</li>
<li>The resulting stream is the 2's complement.</li>
</ol>

<script>
function sample(x){document.getElementById("bin").value=x;process()}
function process(){
 const input=document.getElementById("bin").value.trim();
 const msg=document.getElementById("message");
 if(!input){msg.innerHTML='<p class="error">Please enter a binary stream.</p>';return}
 if(!/^[01]+$/.test(input)){
   msg.innerHTML='<p class="error">❌ Invalid Input: only 0 and 1 are allowed.</p>';
   document.getElementById("resultCard").style.display="none";
   document.getElementById("stepsCard").style.display="none";
   return;
 }
 msg.innerHTML='<p class="success">✓ Valid binary input accepted.</p>';
 let out="",found=false,rows=[];
 let step=1;
 for(let i=input.length-1;i>=0;i--){
   const bit=input[i], before=found?'q1':'q0';
   let ob;
   if(!found){ob=bit;if(bit==='1')found=true}
   else ob=bit==='0'?'1':'0';
   const after=found?'q1':'q0';
   out=ob+out;
   rows.push([step,i+1,bit,before,ob,after]);
   step++;
 }
 document.getElementById("inputOut").textContent=input;
 document.getElementById("outputOut").textContent=out;
 document.getElementById("direction").textContent="✓ Processed from RIGHT → LEFT";
 document.getElementById("resultCard").style.display="block";
 document.getElementById("stepsCard").style.display="block";
 const table=document.getElementById("stepsTable");
 table.innerHTML='<tr><th>Step</th><th>Position</th><th>Read Bit</th><th>State Before</th><th>Output Bit</th><th>State After</th></tr>';
 rows.forEach(r=>{
  table.innerHTML+=`<tr><td>${r[0]}</td><td>${r[1]}</td><td>${r[2]}</td><td>${r[3]}</td><td>${r[4]}</td><td>${r[5]}</td></tr>`;
 });
}
function resetApp(){
 document.getElementById("bin").value="";
 document.getElementById("message").innerHTML="";
 document.getElementById("resultCard").style.display="none";
 document.getElementById("stepsCard").style.display="none";
}
</script>
